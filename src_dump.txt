    mut commands: Commands,
    mut meshes: ResMut<Assets<Mesh>>,
    mut materials: ResMut<Assets<StandardMaterial>>,
) {
    let plane_size = 512.0;
    let wall_height = 512.0;
    let wall_thickness = 1.0;

    // --- РџРѕР» ---
    let floor_mesh = Mesh::from(shape::Plane::from_size(plane_size));
    let floor_collider = collider_from_mesh(&floor_mesh);
    let floor_mesh_handle = meshes.add(floor_mesh);

    let floor_material = materials.add(StandardMaterial {
        base_color: Color::rgb(0.12, 0.18, 0.12),
        perceptual_roughness: 1.0,
        metallic: 0.0,
        ..default()
    });

    commands.spawn((
        PbrBundle {
            mesh: floor_mesh_handle,
            material: floor_material,
            transform: Transform::from_xyz(0.0, 0.0, 0.0),
            ..default()
        },
        RigidBody::Fixed,
        floor_collider,
    ));

    // --- РЎС‚РµРЅС‹ (РўРћР›Р¬РљРћ РљРћР›Р›РђР™Р”Р•Р Р«, Р±РµР· РІРёРґРёРјРѕРіРѕ РјРµС€Р°) ---
    let half_thickness = wall_thickness * 0.5;

    let half_size = plane_size * 0.5;

    // РЎС‚РµРЅС‹ РІРґРѕР»СЊ РѕСЃРё X (РїРѕСЃС‚Р°РІР»РµРЅС‹ РїРѕ РєСЂР°СЏРј РїРѕ Z)

        TransformBundle::from(Transform::from_xyz(
            half_size + half_thickness,
            half_height,
            0.0,
        )),
        RigidBody::Fixed,
        wall_x_collider.clone(),
    ));

    // -X
    commands.spawn((
        TransformBundle::from(Transform::from_xyz(
            -half_size - half_thickness,
            half_height,
            0.0,
        )),
        RigidBody::Fixed,
        wall_x_collider,
    ));

    // РЎС‚РµРЅС‹ РІРґРѕР»СЊ РѕСЃРё Z
    let wall_z_collider = Collider::cuboid(half_size, half_height, half_thickness);

    commands.spawn((
        TransformBundle::from(Transform::from_xyz(
            0.0,
            half_height,
            half_size + half_thickness,
        )),
        RigidBody::Fixed,
        wall_z_collider.clone(),
    ));

    // -Z
    commands.spawn((
        TransformBundle::from(Transform::from_xyz(
            0.0,
            half_height,
            -half_size - half_thickness,
        )),
        RigidBody::Fixed,
        wall_z_collider,
    ));
        // --- РџРѕС‚РѕР»РѕРє (РЅРµРІРёРґРёРјС‹Р№, С‡РёСЃС‚Рѕ РєРѕР»Р»Р°Р№РґРµСЂ) ---

        half_ceiling_thickness, // С‚РѕР»С‰РёРЅР° РїРѕ Y

/// РЎС‚СЂРѕРёРј VoxelGrid 32Г—32Г—32 СЃ С€Р°СЂРѕРј РїРѕ С†РµРЅС‚СЂСѓ.
/// Р­С‚Рѕ Рё РµСЃС‚СЊ "РЅР°РїРѕР»РЅРµРЅРёРµ" DUN РґР»СЏ MVP.
fn build_dun_sphere_grid() -> VoxelGrid {
    // 32Ві РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ
    let mut grid = VoxelGrid::new_default();
    let size = grid.size;

    // Р¦РµРЅС‚СЂ РІ РІРѕРєСЃРµР»СЊРЅС‹С… РєРѕРѕСЂРґРёРЅР°С‚Р°С… (РїРѕ С†РµРЅС‚СЂР°Рј СЏС‡РµРµРє)
    let cx = size.x as f32 / 2.0;
    let cy = size.y as f32 / 2.0;
    let cz = size.z as f32 / 2.0;

    // Р Р°РґРёСѓСЃ вЂ” С‡СѓС‚СЊ РјРµРЅСЊС€Рµ РїРѕР»РѕРІРёРЅС‹ РїРѕ РјРёРЅРёРјР°Р»СЊРЅРѕР№ РѕСЃРё,
    // С‡С‚РѕР±С‹ С€Р°СЂ РЅРµ СѓРїРёСЂР°Р»СЃСЏ РІ РєСЂР°Р№
    let r = (size.x.min(size.y).min(size.z) as f32 * 0.5) - 1.0;
    let r2 = r * r;

    for x in 0..size.x {
        for y in 0..size.y {
            for z in 0..size.z {
                let vx = x as f32 + 0.5;
                let vy = y as f32 + 0.5;
                let vz = z as f32 + 0.5;

                let dx = vx - cx;
                let dy = vy - cy;
                let dz = vz - cz;

                let dist2 = dx * dx + dy * dy + dz * dz;

                if dist2 <= r2 {
                    grid.set(x, y, z, 1);
                }
            }
        }
    }

    grid
}

/// РЎРїР°РІРЅ РѕРґРЅРѕРіРѕ DUN (РєРѕРЅС‚РµР№РЅРµСЂ 32Ві СЃ С€Р°СЂРѕРј РІРЅСѓС‚СЂРё) СЃ Р—РђР”РђРќРќР«Рњ С‚СЂР°РЅСЃС„РѕСЂРјРѕРј.

pub fn spawn_single_dun(

use bevy::prelude::*;
use bevy::render::mesh::{Indices, VertexAttributeValues};
use bevy_rapier3d::prelude::*;

use crate::voxel::grid::VoxelGrid;
use crate::voxel::mesher::VOXEL_SIZE;

pub fn collider_from_mesh(mesh: &Mesh) -> Collider {
    let positions: Vec<Vec3> = match mesh.attribute(Mesh::ATTRIBUTE_POSITION) {
        Some(VertexAttributeValues::Float32x3(verts)) => verts
            .iter()
            .map(|[x, y, z]| Vec3::new(*x, *y, *z))
            .collect(),
        _ => {
            return Collider::cuboid(0.5, 0.5, 0.5);
        }
    };

    let raw_indices: Vec<u32> = match mesh.indices() {
        Some(Indices::U32(ind)) => ind.clone(),
        Some(Indices::U16(ind)) => ind.iter().map(|&i| i as u32).collect(),
        None => (0..positions.len() as u32).collect(),
    };

    let mut indices_triangles = Vec::new();
    for chunk in raw_indices.chunks(3) {
        if let [a, b, c] = chunk {
            indices_triangles.push([*a, *b, *c]);
        }
    }

    if positions.is_empty() || indices_triangles.is_empty() {
        Collider::cuboid(0.5, 0.5, 0.5)
    } else {
        Collider::trimesh(positions, indices_triangles)
    }
}

pub fn collider_from_voxel_grid_box(grid: &VoxelGrid) -> Collider {

    let hx = size.x as f32 * VOXEL_SIZE * 0.5;
    let hy = size.y as f32 * VOXEL_SIZE * 0.5;

}

pub fn build_bevy_mesh(grid: &VoxelGrid) -> Mesh {
    let size = grid.size;

    let extent = Vec3::new(
        size.x as f32 * VOXEL_SIZE,
        size.y as f32 * VOXEL_SIZE,
        size.z as f32 * VOXEL_SIZE,
    );
    let center = extent * 0.5;

    let mut positions: Vec<[f32; 3]> = Vec::new();
    let mut normals:   Vec<[f32; 3]> = Vec::new();
    let mut uvs:       Vec<[f32; 2]> = Vec::new();
    let mut indices:   Vec<u32>      = Vec::new();

    // РҐРµР»РїРµСЂ: РґРѕР±Р°РІР»СЏРµС‚ РѕРґРЅСѓ РєРІР°РґСЂР°С‚РЅСѓСЋ РіСЂР°РЅСЊ (2 С‚СЂРµСѓРіРѕР»СЊРЅРёРєР°).

    let mut push_face = |verts: [Vec3; 4], normal: Vec3| {
        let base = positions.len() as u32;

        for v in verts {
            let p = v - center; // С†РµРЅС‚СЂРёСЂСѓРµРј РІРѕРєСЂСѓРі (0,0,0)
            positions.push(p.to_array());
            normals.push(normal.to_array());
            uvs.push([0.0, 0.0]);
        }

        indices.extend_from_slice(&[
            base,     base + 1, base + 2,
            base,     base + 2, base + 3,
        ]);
    };

    // РЎРѕСЃРµРґ РїСѓСЃС‚РѕР№, РµСЃР»Рё:

           x >= size.x as i32 ||
           y >= size.y as i32 ||
           z >= size.z as i32
        {
            return true;
        }
        grid.get(x as u32, y as u32, z as u32) == 0
    };

    for x in 0..size.x {
        for y in 0..size.y {
            for z in 0..size.z {
                if grid.get(x, y, z) == 0 {

                            Vec3::new(max.x, max.y, max.z),
                            Vec3::new(max.x, min.y, max.z),
                        ],
                        Vec3::X,
                    );
                }

                // -X
                if is_empty(ix - 1, iy, iz) {

                            Vec3::new(min.x, max.y, max.z),
                            Vec3::new(min.x, max.y, min.z),

                if is_empty(ix, iy + 1, iz) {
                    push_face(
                        [
                            // v0: min x, max z
                            Vec3::new(min.x, max.y, max.z),
                            // v1: max x, max z
                            Vec3::new(max.x, max.y, max.z),
                            // v2: max x, min z
                            Vec3::new(max.x, max.y, min.z),
                            // v3: min x, min z

                if is_empty(ix, iy - 1, iz) {

                            // v0: min x, min z

                            // v1: max x, min z
                            Vec3::new(max.x, min.y, min.z),

                if is_empty(ix, iy, iz + 1) {
                    push_face(
                        [
                            // v0: min x, min y
                            Vec3::new(min.x, min.y, max.z),
                            // v1: max x, min y
                            Vec3::new(max.x, min.y, max.z),

        Self {

/// РџРѕРІРѕСЂРѕС‚ РєР°РјРµСЂС‹ РїСЂР°РІРѕР№ РєРЅРѕРїРєРѕР№ РјС‹С€Рё.
pub fn fly_camera_look(
    time: Res<Time>,
    mut mouse_motion: EventReader<MouseMotion>,
    mouse_buttons: Res<ButtonInput<MouseButton>>,
    mut query: Query<(&mut Transform, &mut FlyCamera)>,
) {
    let (mut transform, mut flycam) = if let Ok(v) = query.get_single_mut() {
        v
    } else {
        // СЃСЉРµРґР°РµРј СЃРѕР±С‹С‚РёСЏ, С‡С‚РѕР±С‹ РѕРЅРё РЅРµ РєРѕРїРёР»РёСЃСЊ
        for _ in mouse_motion.read() {}
        return;
    };

    // РєСЂСѓС‚РёРј С‚РѕР»СЊРєРѕ РїСЂРё Р·Р°Р¶Р°С‚РѕР№ РџРљРњ
    if !mouse_buttons.pressed(MouseButton::Right) {
        for _ in mouse_motion.read() {}
        return;
    }

    let mut delta = Vec2::ZERO;
    for ev in mouse_motion.read() {
        delta += ev.delta;
    }
    if delta == Vec2::ZERO {
        return;
    }

    let dt = time.delta_seconds();
    flycam.yaw   -= delta.x * flycam.mouse_sensitivity * dt;
    flycam.pitch -= delta.y * flycam.mouse_sensitivity * dt;

    // РѕРіСЂР°РЅРёС‡РёРІР°РµРј РЅР°РєР»РѕРЅ РІРІРµСЂС…/РІРЅРёР·
    flycam.pitch = flycam.pitch.clamp(-1.5, 1.5);

    transform.rotation =
        Quat::from_axis_angle(Vec3::Y, flycam.yaw) *
        Quat::from_axis_angle(Vec3::X, flycam.pitch);
}

/// Р”РІРёР¶РµРЅРёРµ: WASD + Space/Ctrl, Shift вЂ” СѓСЃРєРѕСЂРµРЅРёРµ.
pub fn fly_camera_move(
    time: Res<Time>,
    keys: Res<ButtonInput<KeyCode>>,
    mut query: Query<(&mut Transform, &FlyCamera)>,
) {
    let (mut transform, flycam) = if let Ok(v) = query.get_single_mut() {
        v
    } else {
        return;
    };

    let mut dir = Vec3::ZERO;

    // РІРїРµСЂС‘Рґ/РЅР°Р·Р°Рґ РїРѕ РЅР°РїСЂР°РІР»РµРЅРёСЋ РєР°РјРµСЂС‹
    if keys.pressed(KeyCode::KeyW) {
        dir += *transform.forward();
    }
    if keys.pressed(KeyCode::KeyS) {
        dir -= *transform.forward();
    }

    // РІР»РµРІРѕ/РІРїСЂР°РІРѕ
    if keys.pressed(KeyCode::KeyA) {
        dir -= *transform.right();
    }
    if keys.pressed(KeyCode::KeyD) {
        dir += *transform.right();
    }

    // РІРІРµСЂС…/РІРЅРёР·
    if keys.pressed(KeyCode::Space) {
        dir += Vec3::Y;
    }
    if keys.pressed(KeyCode::ControlLeft) {
        dir -= Vec3::Y;
    }

    if dir.length_squared() == 0.0 {
        return;
    }

    dir = dir.normalize();
    let mut speed = flycam.move_speed;
    if keys.pressed(KeyCode::ShiftLeft) {
        speed = flycam.move_speed_fast;
    }

    transform.translation += dir * speed * time.delta_seconds();
}

/// Р Р°Р·РјРµСЂ РіСЂРёРґР° РїРѕ РѕРґРЅРѕР№ РѕСЃРё РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ (РјРѕР¶РЅРѕ РјРµРЅСЏС‚СЊ РїСЂРё СЃРѕР·РґР°РЅРёРё)
pub const VOXELS_PER_AXIS: u32 = 32;

/// РџСЂРѕСЃС‚РµР№С€РёР№ РІРѕРєСЃРµР»СЊРЅС‹Р№ РіСЂРёРґ: size Рё РїР»РѕСЃРєРёР№ РјР°СЃСЃРёРІ РґР°РЅРЅС‹С….
/// 0 = РїСѓСЃС‚Рѕ, >0 = РєР°РєРѕР№-С‚Рѕ РјР°С‚РµСЂРёР°Р».
pub struct VoxelGrid {
    pub size: UVec3,
    pub data: Vec<u8>,
}

impl VoxelGrid {
    /// РЎРѕР·РґР°С‘С‚ РіСЂРёРґ СѓРєР°Р·Р°РЅРЅРѕРіРѕ СЂР°Р·РјРµСЂР° Рё Р·Р°РїРѕР»РЅСЏРµС‚ РЅСѓР»СЏРјРё.
    pub fn new(size: UVec3) -> Self {
        let len = (size.x * size.y * size.z) as usize;
        Self {
            size,
            data: vec![0; len],
        }
    }

    /// DUN-РіСЂРёРґ РїРѕ СѓРјРѕР»С‡Р°РЅРёСЋ: 32Г—32Г—32.
    pub fn new_default() -> Self {
        let size = UVec3::splat(VOXELS_PER_AXIS);
        Self::new(size)
    }

    #[inline]
    fn index(&self, x: u32, y: u32, z: u32) -> Option<usize> {
        if x >= self.size.x || y >= self.size.y || z >= self.size.z {
            return None;
        }
        // РїРѕСЂСЏРґРѕРє РєР°Рє Сѓ С‚РµР±СЏ Р±С‹Р»: x + X * (y + Y * z)
        let idx = x + self.size.x * (y + self.size.y * z);
        Some(idx as usize)
    }

    #[inline]
    pub fn get(&self, x: u32, y: u32, z: u32) -> u8 {
        if let Some(i) = self.index(x, y, z) {
            self.data[i]
        } else {
            0
        }
    }

    #[inline]
    pub fn set(&mut self, x: u32, y: u32, z: u32, value: u8) {
        if let Some(i) = self.index(x, y, z) {
            self.data[i] = value;
        }
    }

    /// Р—Р°РїРѕР»РЅРёС‚СЊ РІРµСЃСЊ РѕР±СЉС‘Рј РѕРґРЅРёРј Р·РЅР°С‡РµРЅРёРµРј.
    pub fn fill(&mut self, value: u8) {
        self.data.fill(value);
    }
}

use bevy::prelude::*;
use bevy::render::mesh::{Indices, PrimitiveTopology};
use bevy::render::render_asset::RenderAssetUsages;

use crate::voxel::grid::VoxelGrid;

/// Р Р°Р·РјРµСЂ РѕРґРЅРѕРіРѕ РІРѕРєСЃРµР»СЏ РІ РјРёСЂРѕРІС‹С… РµРґРёРЅРёС†Р°С….
pub const VOXEL_SIZE: f32 = 1.0;

/// РќР°РїСЂР°РІР»РµРЅРёСЏ РґР»СЏ 6 РіСЂР°РЅРµР№ РєСѓР±Р° РІ РёРЅРґРµРєСЃРЅРѕРј РїСЂРѕСЃС‚СЂР°РЅСЃС‚РІРµ РіСЂРёРґР°.
const FACE_OFFSETS: [IVec3; 6] = [
    IVec3::new(0,  1,  0), // +Y (top)
    IVec3::new(0, -1,  0), // -Y (bottom)
    IVec3::new(0,  0,  1), // +Z (front)
    IVec3::new(0,  0, -1), // -Z (back)
    IVec3::new(-1, 0,  0), // -X (left)

/// РќРѕСЂРјР°Р»Рё РґР»СЏ РіСЂР°РЅРµР№ (РїРѕ С‚РµРј Р¶Рµ РёРЅРґРµРєСЃР°Рј).
const FACE_NORMALS: [[f32; 3]; 6] = [
    [0.0,  1.0,  0.0], // +Y
    [0.0, -1.0,  0.0], // -Y
    [0.0,  0.0,  1.0], // +Z
    [0.0,  0.0, -1.0], // -Z
    [-1.0, 0.0,  0.0], // -X
    [1.0,  0.0,  0.0], // +X
];

/// РћРґРёРЅР°РєРѕРІС‹Рµ UV РґР»СЏ РІСЃРµС… РіСЂР°РЅРµР№.
const FACE_UVS: [[f32; 2]; 4] = [
    [0.0, 0.0],
    [1.0, 0.0],
    [1.0, 1.0],
    [0.0, 1.0],
];

/// РЎС‚СЂРѕРёРј РјРµС€ РїРѕ РІРѕРєСЃРµР»СЊРЅРѕР№ СЂРµС€С‘С‚РєРµ:
/// РґР»СЏ РєР°Р¶РґРѕРіРѕ Р·Р°РїРѕР»РЅРµРЅРЅРѕРіРѕ РІРѕРєСЃРµР»СЏ РґРѕР±Р°РІР»СЏРµРј С‚РѕР»СЊРєРѕ С‚Рµ РіСЂР°РЅРё,
/// РіРґРµ СЃРѕСЃРµРґ РїСѓСЃС‚РѕР№ РёР»Рё РІРЅРµ РіСЂР°РЅРёС† РіСЂРёРґР°.
pub fn build_mesh(grid: &VoxelGrid) -> Mesh {
    let mut positions: Vec<[f32; 3]> = Vec::new();
    let mut normals:   Vec<[f32; 3]> = Vec::new();
    let mut uvs:       Vec<[f32; 2]> = Vec::new();
    let mut indices:   Vec<u32>      = Vec::new();

    let size = grid.size;

    for z in 0..size.z {
        for y in 0..size.y {
            for x in 0..size.x {
                if grid.get(x, y, z) == 0 {
                    continue;
                }

                // Р¦РµРЅС‚СЂ РІРѕРєСЃРµР»СЏ РІ Р»РѕРєР°Р»СЊРЅС‹С… РєРѕРѕСЂРґРёРЅР°С‚Р°С…
                let cx = x as f32 + 0.5;
                let cy = y as f32 + 0.5;
                let cz = z as f32 + 0.5;

                let half = VOXEL_SIZE * 0.5;

                let px = cx + half;
                let nx = cx - half;
                let py = cy + half;
                let ny = cy - half;
                let pz = cz + half;
                let nz = cz - half;

                // Р”Р»СЏ РєР°Р¶РґРѕР№ РёР· 6 РіСЂР°РЅРµР№ РїСЂРѕРІРµСЂСЏРµРј СЃРѕСЃРµРґР°
                for face in 0..6 {
                    let offset = FACE_OFFSETS[face];
                    let nx_i = x as i32 + offset.x;
                    let ny_i = y as i32 + offset.y;
                    let nz_i = z as i32 + offset.z;

                    let neighbor_empty = if nx_i < 0
                        || ny_i < 0
                        || nz_i < 0
                        || nx_i >= size.x as i32
                        || ny_i >= size.y as i32
                        || nz_i >= size.z as i32
                    {
                        // Р’С‹С€Р»Рё Р·Р° РіСЂР°РЅРёС†С‹ РіСЂРёРґР° вЂ” СЃС‡РёС‚Р°РµРј, С‡С‚Рѕ С‚Р°Рј РїСѓСЃС‚Рѕ
                        true
                    } else {
                        // Р’РЅСѓС‚СЂРё РіСЂРёРґР° вЂ” РїСЂРѕРІРµСЂСЏРµРј РІРѕРєСЃРµР»СЊ
                        grid.get(nx_i as u32, ny_i as u32, nz_i as u32) == 0
                    };

                    if neighbor_empty {
                        add_face(
                            face,
                            nx, px, ny, py, nz, pz,
                            &mut positions,
                            &mut normals,
                            &mut uvs,
                            &mut indices,
                        );
                    }
                }
            }
        }
    }

    let mut mesh = Mesh::new(
        PrimitiveTopology::TriangleList,
        RenderAssetUsages::default(),
    );

    mesh.insert_attribute(Mesh::ATTRIBUTE_POSITION, positions);
    mesh.insert_attribute(Mesh::ATTRIBUTE_NORMAL, normals);
    mesh.insert_attribute(Mesh::ATTRIBUTE_UV_0, uvs);
    mesh.insert_indices(Indices::U32(indices));

/// Р”РѕР±Р°РІР»СЏРµРј РѕРґРЅСѓ РіСЂР°РЅСЊ РєСѓР±Р° РІ Р±СѓС„РµСЂС‹.
/// face: 0..5 вЂ” РёРЅРґРµРєСЃ РІ FACE_OFFSETS / FACE_NORMALS.
fn add_face(
    face: usize,
    nx: f32,
    px: f32,
    ny: f32,
    py: f32,
    nz: f32,
    pz: f32,
    positions: &mut Vec<[f32; 3]>,
    normals: &mut Vec<[f32; 3]>,
    uvs: &mut Vec<[f32; 2]>,
    indices: &mut Vec<u32>,
) {
    let verts: [[f32; 3]; 4] = match face {
        // +Y (top) вЂ” РЅРѕСЂРјР°Р»СЊ (0, 1, 0)
        0 => [
            [nx, py, pz],
            [px, py, pz],
            [px, py, nz],
            [nx, py, nz],
        ],
        // -Y (bottom) вЂ” РЅРѕСЂРјР°Р»СЊ (0, -1, 0)
        1 => [
            [nx, ny, nz],
            [px, ny, nz],
            [px, ny, pz],
            [nx, ny, pz],
        ],
        // +Z (front)
        2 => [
            [nx, ny, pz],
            [px, ny, pz],
            [px, py, pz],
            [nx, py, pz],
        ],
        // -Z (back)
        3 => [
            [px, ny, nz],
            [nx, ny, nz],
            [nx, py, nz],
            [px, py, nz],
        ],
        // -X (left)
        4 => [
            [nx, ny, nz],
            [nx, ny, pz],
            [nx, py, pz],
            [nx, py, nz],
        ],
        // +X (right)
        5 => [
            [px, ny, pz],
            [px, ny, nz],
            [px, py, nz],
            [px, py, pz],
        ],
        _ => unreachable!(),
    };

    let normal = FACE_NORMALS[face];

    let start = positions.len() as u32;
    for (i, p) in verts.iter().enumerate() {
        positions.push(*p);
        normals.push(normal);
        uvs.push(FACE_UVS[i]);
    }

    indices.extend_from_slice(&[
        start,
        start + 1,
        start + 2,
        start,
        start + 2,
        start + 3,
    ]);
}
